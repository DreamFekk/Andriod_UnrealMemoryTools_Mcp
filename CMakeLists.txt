cmake_minimum_required(VERSION 3.22)
# Toolchain is supplied by build.ps1 or -DCMAKE_TOOLCHAIN_FILE; no host paths here.
project(UnrealMemoryTools)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(UMT_NATIVE_UI "Build the original Android overlay capability" ON)
if(NOT DEFINED CMAKE_RUNTIME_OUTPUT_DIRECTORY)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/outputs/${ANDROID_ABI})
endif()
file(GLOB CORE CONFIGURE_DEPENDS src/UE/*.cpp src/UE/UEGameProfiles/*.cpp src/AutoFix/*.cpp)
file(GLOB KITTY CONFIGURE_DEPENDS KittyMemoryEx/*.cpp)
set(COMMON_UTILS src/Utils/BufferFmt.cpp src/Utils/ProgressUtils.cpp)
add_library(umt_core STATIC ${CORE} ${KITTY} ${COMMON_UTILS} src/Dumper.cpp src/UPackageGenerator.cpp deps/fmt/format.cc)
target_include_directories(umt_core PUBLIC src deps KittyMemoryEx)
target_compile_definitions(umt_core PUBLIC kEXECUTABLE kNO_KEYSTONE)
target_compile_options(umt_core PRIVATE -O3 -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_libraries(umt_core PUBLIC log android z dl)
add_executable(UnrealMemoryTools src/executable.cpp src/App.cpp src/HttpServer.cpp src/FastDump.cpp src/ExplorerApi.cpp)
target_link_libraries(UnrealMemoryTools PRIVATE umt_core)
target_link_options(UnrealMemoryTools PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384 $<$<CONFIG:Release>:-s>)
target_compile_options(UnrealMemoryTools PRIVATE -O3)
add_executable(umt_native_checks EXCLUDE_FROM_ALL tests/native_checks.cpp src/FastDump.cpp)
target_link_libraries(umt_native_checks PRIVATE umt_core)
add_executable(umt_engine_compat_test EXCLUDE_FROM_ALL tests/engine_compat_test.cpp)
target_link_libraries(umt_engine_compat_test PRIVATE umt_core)
add_executable(umt_object_array_layout_test EXCLUDE_FROM_ALL tests/object_array_layout_test.cpp)
target_include_directories(umt_object_array_layout_test PRIVATE src)
if(UMT_NATIVE_UI)
  file(GLOB GUI CONFIGURE_DEPENDS src/GUI/Android_draw/*.cpp src/GUI/Android_Graphics/*.cpp src/GUI/Android_my_imgui/*.cpp src/GUI/Android_touch/*.cpp src/GUI/ImGui/*.cpp src/GUI/ImGui/backends/*.cpp src/GUI/ImGui/misc/freetype/*.cpp src/GUI/My_Utils/*.cpp)
  target_sources(UnrealMemoryTools PRIVATE ${GUI} src/NativeUI.cpp src/SDKExplorer.cpp)
  target_include_directories(UnrealMemoryTools PRIVATE src/GUI src/GUI/Android_draw src/GUI/Android_Graphics src/GUI/Android_my_imgui src/GUI/Android_touch src/GUI/ImGui src/GUI/ImGui/backends src/GUI/ImGui/misc/freetype src/GUI/ImGui/misc/git_freetype src/GUI/My_Utils src/GUI/native_surface)
  target_compile_definitions(UnrealMemoryTools PRIVATE UMT_NATIVE_UI VK_USE_PLATFORM_ANDROID_KHR IMGUI_IMPL_VULKAN_NO_PROTOTYPES IMGUI_ENABLE_FREETYPE IMGUI_DISABLE_DEBUG_TOOLS)
  target_link_libraries(UnrealMemoryTools PRIVATE EGL GLESv3 vulkan ${CMAKE_SOURCE_DIR}/src/GUI/ImGui/misc/git_freetype/${ANDROID_ABI}/libfreetype.a)
endif()
